## 任务 2：添加导出/导入配置功能 (待实现)

### 问题描述

用户每次更新插件版本（删除旧版本 + 加载新版本），Chrome 会分配新的扩展 ID，导致 `chrome.storage.local` 中的配置全部丢失：
- API 配置
- 平台提示词
- AI 提示词（Translate & Polish、Revise）
- 主题设置
- 面板位置

### 解决方案

在设置面板添加"📤 导出配置"和"📥 导入配置"按钮，让用户可以手动备份和恢复配置。

### 配置结构

```javascript
{
  apis: [],                    // API 列表
  activeApiIndex: 0,           // 当前激活的 API 索引
  platforms: {},               // 平台配置 { id: { promptReuse, knowledgeReuse, prompt, knowledge } }
  customPlatforms: [],         // 自定义平台列表
  theme: 'blue',               // 主题
  aiPrompts: {                 // AI 提示词
    translate: { prompt: '' },
    revise: { prompt: '' },
    defaultPlatform: { prompt: '' }
  },
  panelPosition: { x: null, y: null },
  collapsedPosition: { x: null, y: null },
  version: '6.0.0'
}
```

### 实现方案

#### 1. 添加导出/导入函数（content.js）

```javascript
// 导出配置
async function exportConfig() {
  const config = await loadConfig();
  const configJson = JSON.stringify(config, null, 2);
  const blob = new Blob([configJson], { type: 'application/json' });
  const url = URL.createObjectURL(blob);

  const a = document.createElement('a');
  a.href = url;
  a.download = `reply-assistant-config-${new Date().toISOString().slice(0,10)}.json`;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
}

// 导入配置
async function importConfig(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = async (e) => {
      try {
        const importedConfig = JSON.parse(e.target.result);

        // 验证配置格式
        if (!importedConfig.apis || !importedConfig.platforms) {
          throw new Error('Invalid config format');
        }

        // 合并配置到现有配置
        const currentConfig = await loadConfig();
        const mergedConfig = {
          ...currentConfig,
          apis: importedConfig.apis || currentConfig.apis,
          activeApiIndex: importedConfig.activeApiIndex ?? currentConfig.activeApiIndex,
          platforms: importedConfig.platforms || currentConfig.platforms,
          customPlatforms: importedConfig.customPlatforms || currentConfig.customPlatforms,
          theme: importedConfig.theme || currentConfig.theme,
          aiPrompts: importedConfig.aiPrompts || currentConfig.aiPrompts,
          panelPosition: importedConfig.panelPosition || currentConfig.panelPosition,
          collapsedPosition: importedConfig.collapsedPosition || currentConfig.collapsedPosition,
          version: importedConfig.version || currentConfig.version
        };

        await saveConfig(mergedConfig);
        resolve(mergedConfig);
      } catch (error) {
        reject(error);
      }
    };
    reader.onerror = () => reject(reader.error);
    reader.readAsText(file);
  });
}
```

#### 3. 添加事件处理器

```javascript
// 导出配置
document.getElementById('export-config-btn').addEventListener('click', async () => {
  try {
    await exportConfig();
    Progress.success('配置已导出');
  } catch (error) {
    Progress.stop();
    alert('导出失败: ' + error.message);
  }
});

// 导入配置
document.getElementById('import-config-btn').addEventListener('click', () => {
  document.getElementById('import-config-file').click();
});

document.getElementById('import-config-file').addEventListener('change', async (e) => {
  const file = e.target.files[0];
  if (!file) return;

  try {
    Progress.start('正在导入配置...');
    await importConfig(file);
    Progress.success('配置已导入，正在重新加载...');
    setTimeout(() => location.reload(), 1000);
  } catch (error) {
    Progress.stop();
    alert('导入失败: ' + error.message);
  }
  e.target.value = '';  // 清空文件选择
});
```
